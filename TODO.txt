Code structure reference (Claude)

src/
  types.ts              # Task, TaskInstance, Priority, RecurrenceRule
  store.ts              # in-memory state + CRUD + change notification
  dates.ts              # pure date math: getMonthGrid, getWeekDays, addDays...
  recurrence.ts         # expand a RecurrenceRule into concrete instances
  busyness.ts           # calculateBusyness(date, instances) -> 0-1

  render/
    month-view.ts
    week-view.ts
    day-view.ts
    dom-helpers.ts       # small createElement-style utilities

  ui/
    view-switcher.ts
    task-view-modal.ts
    task-edit-modal.ts   # the mode/fieldset logic we've been discussing

  main.ts                # wires it all up, initial render, top-level listeners
